
def link(s:readbuf, o:mutbuf, w:mutbuf) -> u32:
    if length(w) < 16777216:
        return 4294967294
    if length(s) < 32 or length(s) > 8388608:
        return 4294967295
    if uint(s, 0, 8) != 3544424472276715848 or uint(s, 24, 8) != 0:
        return 4294967295
    count = uint(s, 8, 8)
    name = uint(s, 16, 8)
    if count == 0 or count > 64 or name == 0 or name > 4096 or name > length(s) - 32:
        return 4294967295
    if name != 4 or uint(s, 32, 4) != 1885697139:
        return 4294967295
    fill_range(w, 0, 256, 0)
    put(w, 0, count)
    put(w, 40, 4096)
    put(w, 72, 32)
    put(w, 80, name)
    if objects(s, w) == 0:
        return 4294967295
    data = aligned(get(w, 40), 4096)
    if data > 8388608 or get(w, 48) > 8388608 - data:
        return 4294967294
    total = data + get(w, 48)
    if length(o) < total:
        return 4294967294
    put(w, 56, data)
    put(w, 64, total)
    i = 0
    while i < get(w, 8):
        sec = 16384 + i * 48
        if (get(w, sec + 16) & 3) == 3:
            put(w, sec + 24, get(w, sec + 24) + data)
        i = i + 1
    if symbols(s, w) == 0:
        return 4294967295
    if module_contracts(s, w) == 0:
        return 4294967295
    if stack_contracts(s, w) == 0:
        return 4294967295
    found = global_find(s, w, 32, name)
    if found == 0:
        return 4294967295
    record = get(w, found + 16)
    if module_entry(s, w, record) == 0:
        return 4294967295
    flags = get(w, record + 24)
    if ((flags >> 8) & 255) != 2 or ((flags >> 16) & 255) != 1 or ((flags >> 24) & 4) == 0:
        return 4294967295
    put(w, 88, get(w, record + 16) - 4194304)
    fill_range(w, 3145728, 8388608, 0)
    if relocations(s, w) == 0:
        return 4294967295
    fill_range(o, 0, total, 0)
    headers(o, w)
    i = 0
    while i < get(w, 8):
        sec = 16384 + i * 48
        if (get(w, sec + 16) & 2) != 0 and get(w, sec + 32) == 1:
            copy_bytes(s, o, get(w, sec), get(w, sec + 24), get(w, sec + 8))
        i = i + 1
    i = 0
    while i < get(w, 32):
        patch = 1572864 + i * 24
        write(o, get(w, patch), get(w, patch + 16), get(w, patch + 8))
        i = i + 1
    return total

def objects(s:readbuf, w:mutbuf) -> u64:
    pos = 32 + get(w, 80)
    obj = 0
    while obj < get(w, 0):
        if within(length(s), pos, 8) == 0:
            return 0
        size = uint(s, pos, 8)
        pos = pos + 8
        if size < 64 or within(length(s), pos, size) == 0:
            return 0
        if uint(s, pos, 7) != 282584257676671 or (byte(s, pos + 7) != 0 and byte(s, pos + 7) != 3) or byte(s, pos + 8) != 0:
            return 0
        if uint(s, pos + 16, 8) != 4299030529 or uint(s, pos + 24, 8) != 0 or uint(s, pos + 32, 8) != 0:
            return 0
        if uint(s, pos + 48, 4) != 0 or uint(s, pos + 52, 2) != 64 or uint(s, pos + 56, 2) != 0 or uint(s, pos + 58, 2) != 64:
            return 0
        table = uint(s, pos + 40, 8)
        count = uint(s, pos + 60, 2)
        strings = uint(s, pos + 62, 2)
        if count == 0 or count > 4096 - get(w, 8) or strings >= count or table < 64 or within(size, table, count * 64) == 0:
            return 0
        record = 4096 + obj * 80
        fill_range(w, record, 80, 0)
        put(w, record, pos)
        put(w, record + 8, size)
        put(w, record + 16, get(w, 8))
        put(w, record + 24, count)
        put(w, record + 72, count)
        i = 0
        while i < count:
            h = pos + table + i * 64
            kind = uint(s, h + 4, 4)
            flags = uint(s, h + 8, 8)
            offset = uint(s, h + 24, 8)
            extent = uint(s, h + 32, 8)
            align = uint(s, h + 48, 8)
            if i == 0:
                j = 0
                while j < 64:
                    if byte(s, h + j) != 0:
                        return 0
                    j = j + 1
            if kind != 8 and within(size, offset, extent) == 0:
                return 0
            if align == 0:
                align = 1
            if align > 4096 or (align & (align - 1)) != 0:
                return 0
            # Our startup has neither CET feature. GNU FEATURE_1_AND therefore
            # intersects to zero, just as ld with this startup object. Admit
            # only that understood property; never discard a needed/unknown
            # property or arbitrary allocated note.
            if kind == 7 and (flags & 2) != 0:
                if flags != 2 or property_note(s, pos + offset, extent) == 0:
                    return 0
                flags = 0
            sec = 16384 + (get(w, 8) + i) * 48
            put(w, sec, pos + offset)
            put(w, sec + 8, extent)
            put(w, sec + 16, flags)
            put(w, sec + 24, 0)
            put(w, sec + 32, kind)
            put(w, sec + 40, align)
            if (flags & 2) != 0:
                if (kind != 1 and kind != 8) or (flags & 18446744073709551608) != 0 or (flags & 5) == 5 or (kind == 8 and (flags & 4) != 0):
                    return 0
                field = 40
                if (flags & 1) != 0:
                    field = 48
                out = aligned(get(w, field), align)
                if out > 8388608 or extent > 8388608 - out:
                    return 0
                put(w, sec + 24, out)
                put(w, field, out + extent)
            if kind == 2:
                if get(w, record + 72) != count or flags != 0 or uint(s, h + 56, 8) != 24 or extent == 0 or extent % 24 != 0:
                    return 0
                link = uint(s, h + 40, 4)
                if link >= count:
                    return 0
                sh = pos + table + link * 64
                str_at = uint(s, sh + 24, 8)
                str_size = uint(s, sh + 32, 8)
                if uint(s, sh + 4, 4) != 3 or uint(s, sh + 8, 8) != 0 or str_size == 0 or within(size, str_at, str_size) == 0 or byte(s, pos + str_at) != 0:
                    return 0
                if extent // 24 > 32768 - get(w, 16) or uint(s, h + 44, 4) > extent // 24:
                    return 0
                put(w, record + 32, h)
                put(w, record + 40, extent // 24)
                put(w, record + 48, pos + str_at)
                put(w, record + 56, str_size)
                put(w, record + 64, get(w, 16))
                put(w, record + 72, i)
                put(w, 16, get(w, 16) + extent // 24)
            if kind == 9 and extent != 0:
                return 0
            i = i + 1
        if get(w, record + 72) == count:
            return 0
        put(w, 8, get(w, 8) + count)
        pos = pos + size
        obj = obj + 1
    if pos != length(s):
        return 0
    return 1

def symbols(s:readbuf, w:mutbuf) -> u64:
    obj = 0
    while obj < get(w, 0):
        record = 4096 + obj * 80
        i = 0
        while i < get(w, record + 40):
            if read_symbol(s, w, record, i) == 0:
                return 0
            i = i + 1
        obj = obj + 1
    return 1

def read_symbol(s:readbuf, w:mutbuf, record:u64, i:u64) -> u64:
    h = get(w, record + 32)
    table = get(w, record) + uint(s, h + 24, 8)
    local = uint(s, h + 44, 4)
    at = table + i * 24
    sym = 262144 + (get(w, record + 64) + i) * 32
    name = uint(s, at, 4)
    binding = byte(s, at + 4) >> 4
    kind = byte(s, at + 4) & 15
    section = uint(s, at + 6, 2)
    value = uint(s, at + 8, 8)
    extent = uint(s, at + 16, 8)
    if (binding != 0 and binding != 1 and binding != 2) or kind > 4 or byte(s, at + 5) > 3:
        return 0
    if (i < local and binding != 0) or (i >= local and binding == 0):
        return 0
    n = string_size(s, get(w, record + 48), get(w, record + 56), name)
    if n == 4294967295 or (binding != 0 and n == 0):
        return 0
    put(w, sym, get(w, record + 48) + name)
    put(w, sym + 8, n)
    defined = 0
    flags = 0
    if i == 0:
        j = 0
        while j < 24:
            if byte(s, at + j) != 0:
                return 0
            j = j + 1
    if section == 0:
        if value != 0 or extent != 0:
            return 0
    else:
        if section == 65521:
            defined = 2
        else:
            if section >= get(w, record + 24):
                return 0
            sec = 16384 + (get(w, record + 16) + section) * 48
            flags = get(w, sec + 16)
            if (flags & 2) == 0:
                # File/debug-only symbols have no runtime address;
                # record them as undefined, refusing any actual use.
                defined = 0
            else:
                if value > get(w, sec + 8) or extent > get(w, sec + 8) - value:
                    return 0
                defined = 1
                value = 4194304 + get(w, sec + 24) + value
    put(w, sym + 16, value)
    put(w, sym + 24, binding | (kind << 8) | (defined << 16) | (flags << 24))
    if binding != 0 and defined != 0:
        old = global_find(s, w, get(w, sym), n)
        if old != 0:
            previous = get(w, old + 24)
            if previous == 1 and binding == 1:
                return 0
            if previous == 2 and binding == 1:
                put(w, old + 16, sym)
                put(w, old + 24, binding)
        else:
            if get(w, 24) >= 8192:
                return 0
            old = 1310720 + get(w, 24) * 32
            put(w, old, get(w, sym))
            put(w, old + 8, n)
            put(w, old + 16, sym)
            put(w, old + 24, binding)
            put(w, 24, get(w, 24) + 1)
    return 1

def relocations(s:readbuf, w:mutbuf) -> u64:
    obj = 0
    while obj < get(w, 0):
        record = 4096 + obj * 80
        table = get(w, record) + uint(s, get(w, record) + 40, 8)
        i = 0
        while i < get(w, record + 24):
            h = table + i * 64
            if uint(s, h + 4, 4) == 4:
                count = uint(s, h + 32, 8)
                target = uint(s, h + 44, 4)
                if uint(s, h + 56, 8) != 24 or count % 24 != 0 or uint(s, h + 40, 4) != get(w, record + 72) or target >= get(w, record + 24):
                    return 0
                sec = 16384 + (get(w, record + 16) + target) * 48
                if (get(w, sec + 16) & 2) == 0:
                    if count != 0:
                        return 0
                j = 0
                while j < count:
                    at = get(w, record) + uint(s, h + 24, 8) + j
                    if relocate_one(s, w, record, sec, at) == 0:
                        return 0
                    j = j + 24
            i = i + 1
        obj = obj + 1
    return 1

def relocate_one(s:readbuf, w:mutbuf, record:u64, sec:u64, at:u64) -> u64:
    offset = uint(s, at, 8)
    kind = uint(s, at + 8, 4)
    symbol = uint(s, at + 12, 4)
    addend = uint(s, at + 16, 8)
    if kind != 0:
        width = 4
        if kind == 1:
            width = 8
        else:
            if kind != 2 and kind != 4 and kind != 10 and kind != 11:
                return 0
        if within(get(w, sec + 8), offset, width) == 0 or symbol >= get(w, record + 40):
            return 0
        sym = 262144 + (get(w, record + 64) + symbol) * 32
        info = get(w, sym + 24)
        address = get(w, sym + 16)
        if (info & 255) != 0:
            found = global_find(s, w, get(w, sym), get(w, sym + 8))
            if found == 0:
                if (info & 255) != 2:
                    return 0
                address = 0
            else:
                address = get(w, get(w, found + 16) + 16)
        else:
            if ((info >> 16) & 255) == 0:
                if symbol != 0:
                    return 0
                address = 0
        patch = get(w, sec + 24) + offset
        value = address + addend
        if kind == 2 or kind == 4:
            value = value - (4194304 + patch)
        if kind == 10 and value > 4294967295:
            return 0
        if (kind == 2 or kind == 4 or kind == 11) and value > 2147483647 and value < 18446744071562067968:
            return 0
        if get(w, 32) >= 32768:
            return 0
        k = 0
        while k < width:
            if byte(w, 3145728 + patch + k) != 0:
                return 0
            store(w, 3145728 + patch + k, 1)
            k = k + 1
        p = 1572864 + get(w, 32) * 24
        put(w, p, patch)
        put(w, p + 8, width)
        put(w, p + 16, value)
        put(w, 32, get(w, 32) + 1)
    return 1

def property_note(s:readbuf, at:u64, size:u64) -> u64:

def global_find(s:readbuf, w:mutbuf, at:u64, size:u64) -> u64:

def string_size(s:readbuf, at:u64, size:u64, offset:u64) -> u64:
    if offset >= size:
        return 4294967295
    i = 0
    while i < size - offset and i <= 4096:

            return i
        i = i + 1
    return 4294967295

def within(size:u64, at:u64, count:u64) -> u64:

def aligned(value:u64, align:u64) -> u64:
    return (value + align - 1) & (0 - align)

def uint(s:readbuf, at:u64, size:u64) -> u64:
    value = 0
    i = 0
    while i < size:
        value = value | (byte(s, at + i) << (i * 8))
        i = i + 1
    return value

def get(w:mutbuf, at:u64) -> u64:
    value = 0
    i = 0
    while i < 8:
        value = value | (byte(w, at + i) << (i * 8))
        i = i + 1
    return value

def write(o:mutbuf, at:u64, value:u64, size:u64) -> u64:
    i = 0
    while i < size:
        store(o, at + i, (value >> (i * 8)) & 255)
        i = i + 1
    return 0

def put(w:mutbuf, at:u64, value:u64) -> u64:
    return write(w, at, value, 8)

def stack_unit_cost(s:readbuf, w:mutbuf, object:u64) -> u64:

    count = uint(s, at + 16, 8)
    width = uint(s, at + 40, 8)
    matrix = at + 64 + count * 64
    costs = 13112832 + object * 2048
    record = 4096 + object * 80
    fill_range(w, 13244416, 2048, 0)
    i = 0
    while i < count:
        function = at + 64 + i * 64
        if uint(s, function + 40, 8) == 1:
            symbol = uint(s, function + 32, 8)
            sym = 262144 + (get(w, record + 64) + symbol) * 32
            found = global_find(s, w, get(w, sym), get(w, sym + 8))
            if found == 0:
                return 0
            index = (get(w, found + 16) - 262144) // 32
            target = get(w, 12582912 + index * 8)
            provider = get(w, 12845056 + index * 8)
            if target == 0 or provider == 0 or provider > get(w, 0):
                return 0
            ordinal = uint(s, target + 192, 8)
            if ordinal >= uint(s, get(w, 13112320 + (provider - 1) * 8) + 16, 8):
                return 0
            value = get(w, 13112832 + (provider - 1) * 2048 + ordinal * 8)
            if value == 0:
                return 0
            put(w, costs + i * 8, value)
            put(w, 13244416 + i * 8, 2)
        i = i + 1
    i = 0
    while i < count:
        if get(w, 13244416 + i * 8) == 0:
            depth = 1
            put(w, 13246464, i)
            put(w, 13246472, 0)
            put(w, 13246480, 0)
            put(w, 13244416 + i * 8, 1)
            while depth > 0:
                cursor = 13246464 + (depth - 1) * 24
                current = get(w, cursor)
                next = get(w, cursor + 8)
                best = get(w, cursor + 16)
                if next == count:
                    value = uint(s, at + 64 + current * 64 + 24, 8) + 16
                    if value > 8384512 or best > 8384512 - value:
                        return 0
                    value = value + best
                    put(w, costs + current * 8, value)
                    put(w, 13244416 + current * 8, 2)
                    if value > get(w, 104):
                        put(w, 104, value)
                    depth = depth - 1
                    if depth > 0:
                        parent = 13246464 + (depth - 1) * 24
                        if value > get(w, parent + 16):
                            put(w, parent + 16, value)
                else:
                    put(w, cursor + 8, next + 1)
                    bits = byte(s, matrix + current * width + next // 8)
                    if (bits & (1 << (next % 8))) != 0:
                        color = get(w, 13244416 + next * 8)
                        if color == 1:
                            return 0
                        if color == 2:
                            value = get(w, costs + next * 8)
                            if value > best:
                                put(w, cursor + 16, value)
                        else:
                            if depth >= count:
                                return 0
                            child = 13246464 + depth * 24
                            put(w, child, next)
                            put(w, child + 8, 0)
                            put(w, child + 16, 0)
                            put(w, 13244416 + next * 8, 1)
                            depth = depth + 1
        i = i + 1
    return 1

def stack_modrm_size(s:readbuf, at:u64, end:u64) -> u64:
    if at >= end:
        return 0
    modrm = byte(s, at)
    mode = modrm >> 6
    rm = modrm & 7
    size = 1
    if mode != 3:
        if rm == 4:
            if at + size >= end:
                return 0
            base = byte(s, at + size) & 7
            size = size + 1
            if mode == 0 and base == 5:
                size = size + 4
        if mode == 0 and rm == 5:
            size = size + 4
        if mode == 1:
            size = size + 1
        if mode == 2:
            size = size + 4
    if size > end - at:
        return 0
    return size

def stack_instruction(s:readbuf, at:u64, end:u64) -> u64:
    if at >= end:
        return 0
    pos = at
    rex = 0
    opcode = byte(s, pos)
    if opcode >= 64 and opcode <= 79:
        rex = opcode
        pos = pos + 1
        if pos >= end:
            return 0
        opcode = byte(s, pos)
    pos = pos + 1
    kind = 0
    immediate = 0
    modrm = 0
    if opcode == 232:
        if rex != 0:
            return 0
        immediate = 4
        kind = 1
    else:
        if opcode == 233:
            immediate = 4
            kind = 2
        else:
            if opcode == 235 or (opcode >= 112 and opcode <= 127):
                immediate = 1
                kind = 3
            else:
                if opcode >= 184 and opcode <= 191:
                    immediate = 4
                    if (rex & 8) != 0:
                        immediate = 8
                else:
                    if opcode == 15:
                        if pos >= end:
                            return 0
                        second = byte(s, pos)
                        pos = pos + 1
                        if second >= 128 and second <= 143:
                            immediate = 4
                            kind = 2
                        else:
                            if second == 175 or second == 182 or second == 190 or second == 191 or (second >= 64 and second <= 79) or (second >= 144 and second <= 159):
                                modrm = 1
                            else:
                                if second != 11:
                                    return 0
                    else:
                        if opcode == 1 or opcode == 9 or opcode == 33 or opcode == 41 or opcode == 49 or opcode == 57 or opcode == 99 or opcode == 133 or opcode == 136 or opcode == 137 or opcode == 138 or opcode == 139 or opcode == 141 or opcode == 211:
                            modrm = 1
                        else:
                            if opcode == 129:
                                modrm = 1
                                immediate = 4
                            else:
                                if opcode == 131 or opcode == 193:
                                    modrm = 1
                                    immediate = 1
                                else:
                                    if opcode == 247:
                                        if pos >= end or (byte(s, pos) & 56) < 16:
                                            return 0
                                        modrm = 1
                                    else:
                                        if opcode == 37:
                                            immediate = 4
                                        else:
                                            if opcode == 243:
                                                if rex != 0 or pos >= end:
                                                    return 0
                                                second = byte(s, pos)
                                                if second != 164 and second != 170:
                                                    return 0
                                                pos = pos + 1
                                            else:
                                                if opcode >= 80 and opcode <= 95:
                                                    if opcode == 85 and (rex & 1) == 0:
                                                        kind = 4
                                                else:
                                                    if opcode != 144 and opcode != 153 and opcode != 195 and opcode != 201 and opcode != 252 and opcode != 253:
                                                        return 0
    if modrm != 0:
        size = stack_modrm_size(s, pos, end)
        if size == 0:
            return 0
        pos = pos + size
    if immediate > end - pos:
        return 0
    pos = pos + immediate
    if pos - at > 15:
        return 0
    return (kind << 8) | (pos - at)

def stack_relative(s:readbuf, absolute:u64, operand:u64, size:u64) -> u64:
    value = uint(s, absolute, size)
    next = operand + size
    sign = 1 << (size * 8 - 1)
    if value >= sign:
        amount = sign * 2 - value
        if amount > next:
            return 18446744073709551615
        return next - amount
    return next + value

def stack_relocation(s:readbuf, at:u64, count:u64, offset:u64) -> u64:
    low = 0
    high = count
    while low < high:
        middle = low + (high - low) // 2
        item = at + middle * 24
        value = uint(s, item, 8)
        if value == offset:
            return item
        if value < offset:
            low = middle + 1
        else:
            high = middle
    return 0

def stack_target_function(s:readbuf, w:mutbuf, offset:u64) -> u64:

    high = get(w, 13393968)

        ordinal = get(w, 13391872 + middle * 8)
        value = uint(s, at + 64 + ordinal * 64 + 8, 8)
        if value == offset:
            return ordinal
        if value < offset:
            low = middle + 1
        else:
            high = middle
    return 18446744073709551615

def stack_bind_call(s:readbuf, w:mutbuf, caller:u64, operand:u64) -> u64:
    at = get(w, 13393920)
    text = get(w, 13393928)
    relocation = stack_relocation(s, get(w, 13393944), get(w, 13393952), operand)
    target = 18446744073709551615
    if relocation != 0:
        symbol = uint(s, relocation + 12, 4)
        contract = get(w, 13393960) + 64 + (symbol - 2) * 256
        target = uint(s, contract + 192, 8)
        if uint(s, at + 64 + target * 64 + 40, 8) != 1:
            return 0
        put(w, 13394008, get(w, 13394008) + 1)
    else:
        offset = stack_relative(s, text + operand, operand, 4)
        if offset >= get(w, 13393936):
            return 0
        if offset >= get(w, 13393976):
            if offset == get(w, 13393984) or offset == get(w, 13393992) or offset == get(w, 13394000):
                return 1
            return 0
        target = stack_target_function(s, w, offset)
    if target >= uint(s, at + 16, 8) or target == caller:
        return 0
    position = 13252608 + caller * uint(s, at + 40, 8) + target // 8
    store(w, position, byte(w, position) | (1 << (target % 8)))
    return 1

def stack_scan_function(s:readbuf, w:mutbuf, ordinal:u64, phase:u64) -> u64:
    at = get(w, 13393920)
    function = at + 64 + ordinal * 64
    start = uint(s, function + 8, 8)
    end = start + uint(s, function + 16, 8)
    text = get(w, 13393928)
    body = start + 4 + ((uint(s, function + 24, 8) + 4095) // 4096) * 11
    pos = start
    while pos < end:
        decoded = stack_instruction(s, text + pos, text + end)
        size = decoded & 255
        kind = decoded >> 8
        if size == 0:
            return 0
        if kind == 4 and pos != start:
            return 0
        if phase == 0:
            boundary = 13260800 + pos // 8
            store(w, boundary, byte(w, boundary) | (1 << (pos % 8)))
            if kind == 1:
                if stack_bind_call(s, w, ordinal, pos + size - 4) == 0:
                    return 0
        else:
            if kind == 2 or kind == 3:
                width = 4
                if kind == 3:
                    width = 1
                operand = pos + size - width
                target = stack_relative(s, text + operand, operand, width)
                if target < body or target >= end:
                    return 0
                if (byte(w, 13260800 + target // 8) & (1 << (target % 8))) == 0:
                    return 0
        pos = pos + size
    return 1
